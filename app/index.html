import unittest from vcore import ExecutionController, ShotAnalyzer from vcore.board import BoardState, Coin class TestExecution(unittest.TestCase): def test_execution_feedback(self): board = BoardState( striker=Coin(id="striker", x=120.0, y=300.0, radius=18.0, color="white", is_striker=True), coins=[ Coin(id="c1", x=290.0, y=300.0, radius=12.0, color="black"), Coin(id="c2", x=330.0, y=320.0, radius=12.0, color="red"), ], ) shot = ShotAnalyzer.select_best_shot(board) controller = ExecutionController() result = controller.execute(board, shot) self.assertIsNotNone(result) self.assertTrue(result.turn >= 1) self.assertGreaterEqual(len(result.log), 1) if __name__ == "__main__": unittest.main()
